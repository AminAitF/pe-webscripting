Gefixte code:
javascriptconst prompt = require('prompt-sync')();

let getal = Number(prompt('Geef een nummer: '));

function fibonacci(n) {
    if (n < 0) return "Geef een positief getal!";
    if (n === 0) return 0;
    if (n === 1) return 1;

    return fibonacci(n - 2) + fibonacci(n - 1);
}

console.log(fibonacci(getal));
