Het probleem:
i = a.length → dat is index 5, maar arrays gaan van 0 tot 4!
Dus eerste push is a[5] = undefined 😬
Fix:

Werkende code:
javascriptfunction reverseArray(...args) {
    let a = [...args];
    let b = [];

    for (let i = a.length - 1; i >= 0; i--) {
        b.push(a[i]);
    }
    return b;
}
console.log(reverseArray(1,2,3,4,5));
// Output: [5, 4, 3, 2, 1]
